from django.shortcuts import render

def branches_list(request):
    return render(request, 'branches_list.html', {'title': 'Філії'})

class Branch: # потім цей тип переїде в models.py, а поки що це просто клас для передачі даних у шаблон
    """Простий клас, що представляє філію (імітація моделі без БД)"""

    def __init__(self, city, is_main=False, address="", phone_main="", phone_short="", email="", description=""):
        self.city = city
        self.is_main = is_main
        self.address = address
        self.phone_main = phone_main
        self.phone_short = phone_short
        self.email = email
        self.description = description

    def __str__(self):
        status = "головна" if self.is_main else "регіональна"
        return f"{self.city} ({status})"

def odesa(request):
